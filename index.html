▸ One-time database setup SQL (run once in Supabase → SQL Editor) 
create table if not exists recon_debits (
  id          bigint generated always as identity primary key,
  ref         text not null unique,
  vendor      text,
  amount      numeric,
  paid_date   date,
  created_at  timestamptz default now()
);

create table if not exists recon_credits (
  id            bigint generated always as identity primary key,
  ref           text not null unique,
  amount        numeric,
  received_date date,
  created_at    timestamptz default now()
);

-- allow the anon key to read/write (simple setup for a private tool)
alter table recon_debits  enable row level security;
alter table recon_credits enable row level security;
create policy "anon all recon_debits"  on recon_debits  for all using (true) with check (true);
create policy "anon all recon_credits" on recon_credits for all using (true) with check (true);
